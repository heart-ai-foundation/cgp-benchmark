def favorite_item(user_id: int, item_id: int) -> dict:
    # Find the user
    user = None
    for u in USERS:
        if u.id == user_id:
            user = u
            break
    
    if user is None:
        return {"error": "User not found"}
    
    # Check if user owns the item
    if not user.owns_item(item_id):
        return {"error": "Cannot favorite item not owned by user"}
    
    # Check if item exists
    item_exists = any(item.id == item_id for item in ITEMS)
    if not item_exists:
        return {"error": "Item not found"}
    
    # Add to favorites if not already there (idempotent)
    if item_id not in user.favorite_item_ids:
        user.favorite_item_ids.append(item_id)
    
    return {"success": True, "message": "Item favorited"}

def test_favorite_item_success():

    assert result["success"] is True

def test_favorite_item_non_owned_fails():
    result = favorite_item(1, 3)  # User 1 tries to favorite item 3 (owned by user 2)

    assert "Cannot favorite item not owned by user" in result["error"]


def test_favorite_item_idempotent():
    # Favorite the same item twice
    result1 = favorite_item(1, 1)
    result2 = favorite_item(1, 1)
    assert result1["success"] is True
    assert result2["success"] is True
    
    # Check that it's only in favorites once
    favorites = list_favorite_items(1)
    favorite_ids = [item["id"] for item in favorites]
    assert favorite_ids.count(1) == 1
