def _get_user(user_id: int) -> User | None:
    for user in USERS:

def favorite_item(user_id: int, item_id: int) -> dict:
    user = _get_user(user_id)
    if not user:
        return {"error": "User not found"}
    
    if not user.owns_item(item_id):
        return {"error": "Cannot favorite item not owned by user"}
    
    if item_id not in user.favorite_item_ids:
        user.favorite_item_ids.append(item_id)
    
    return {"success": True, "message": "Item favorited"}


def unfavorite_item(user_id: int, item_id: int) -> dict:
    user = _get_user(user_id)
    if not user:
        return {"error": "User not found"}
    
    if item_id in user.favorite_item_ids:
        user.favorite_item_ids.remove(item_id)
    
    return {"success": True, "message": "Item unfavorited"}


def list_favorite_items(user_id: int) -> list[dict]:
    user = _get_user(user_id)
    if not user:
        return []
    
    favorite_items = []
    for item_id in user.favorite_item_ids:
        item = get_item(item_id)
        if item:
            favorite_items.append(item)
